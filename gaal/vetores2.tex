\section{Produto Escalar}

\subsection*{O Produto Escalar}

\begin{frame}[label=vetores2]{O produto escalar}

\begin{defin}
Dados dois vetores $\vec{u}=(u_1,u_2,\ldots,u_n)$ e $\vec{v}=(v_1,v_2,\ldots,v_n)$ do $\R^n$, definimos o {\color{blue} produto escalar (ou produto interno)} entre eles por:
\[\vec{u}\cdot \vec{v}=u_1v_1+u_2v_2+\cdots+ u_nv_n.\]
\end{defin}

\begin{exe}
Calcule $\vec{u}\cdot \vec{v}$, em que $\vec{u}=(1,2,-3)$ e $\vec{v}=(-3,5,2)$.
\end{exe}

\end{frame}

\begin{frame}[label=vetores2]

\begin{block}{Propriedades do produto interno} Para quaisquer vetores $\vu$, $\vv$ e $\vw$ em $\R^n$ e para qualquer número real $\lambda$, valem as propriedades:
			
			\begin{enumerate}
				\item ${\vu}\cdot {\vv}={\vv}\cdot{\vu}$;
				\item $\lambda( {\vu}\cdot {\vv})=(\lambda  {\vu})\cdot {\vv}=  {\vu}\cdot(\lambda {\vv});$
				\item $( {\vu}+ \vv)\cdot {\vw}= {\vu}\cdot {\vw}+ {\vv}\cdot {\vw}$;
				\item $ {\vu}\cdot {\vu}=\n{ {\vu}}^2$.
%				\item $| {u}\cdot {v}|\leq \n{\vect{u}}\n{\vect{v}}$ (desigualdade de Cauchy-Schwartz)
			\end{enumerate}            
		\end{block}
\begin{exe}
\begin{enumerate}
\item Calcule $\|u\|$
\end{enumerate}


Mostre que
\[\|\vu-\vv\|^2=\|\vu\|^2-2\vu\cdot \vv+\|\vv\|^2.\]
\end{exe}

\end{frame}

\subsection*{Ângulo entre vetores}
\begin{frame}[label=vetores2]{Ângulo entre vetores no $\R^2$ e $\R^3$}

		\end{defin}

\subsection*{Lei dos Cossenos}
\begin{frame}[label=vetores2]


	\begin{exampleblock}{Lei dos Cossenos}
	Seja $ABC$ um triângulo qualquer com lados ${\color{red}a}, {\color{blue}b}$ e ${\color{blue}c}$, então
	\[{\color{red}a^2}={\color{blue}b^2}+{\color{blue}c^2}-2{\color{blue}b c}\cos( {\color{red}\hat{A}}),\]
	onde ${\color{red}\hat{A}}$ é o ângulo oposto ao lado $a$.

\begin{frame}[label=vetores2]{Generalização de Ângulo entre vetores}

\begin{defin}
Dados vetores $\vu$ e $\vv$ em $\R^n$, definimos o {\color{blue}ângulo entre $\vu$ e $\vv$} como sendo o valor $0\leq \theta\leq \pi$ que satisfaz:
\[\cos(\theta)=\frac{\vu\cdot\vv}{\|\vu\|\, \|\vv\|}. \] 
\end{defin}

\end{frame}

\subsection*{Produto escalar usando o sympy}
\begin{frame}[label=vetores2,fragile=singleslide]{Produto escalar usando o sympy}

	\begin{footnotesize}
\begin{pyverbatim}
import sympy as sp
import mpmath

u=sp.Matrix([1.12,-3.25,2.07,-1.83 ])
v=sp.Matrix([-2.29,1.72,4.33,-1.54])
uv=u.dot(v)
nu=u.norm()
nv=v.norm()
cos=uv/(nu*nv)
theta=sp.acos(cos)
thetag=round(mp.degrees(theta),2)
\end{pyverbatim}
	\end{footnotesize}	

\begin{pycode}
import sympy as sp
import mpmath as mp

u=sp.Matrix([1.12,-3.25,2.07,-1.83 ])
v=sp.Matrix([-2.29,1.72,4.33,-1.54])
uv=u.dot(v)
nu=u.norm()
nv=v.norm()
cos=uv/(nu*nv)
theta=sp.acos(cos)
thetag=round(mp.degrees(theta),2)
\end{pycode}
Denotando-se os vetores como matrizes colunas, temos:
\[u\cdot v=\pyl{u}\cdot \pyl{v}=\pyl{uv}\]
\[\cos(\theta)\approx\pyl{cos}\Rightarrow \theta\approx\pyl{theta} \text{ rad}\approx \pyl{thetag}^\circ.\]
\end{frame}
