\subsection*{Combinação Linear}
\begin{frame}[label=lild,fragile]{Combinação Linear}
\begin{pycode} 
import sympy as sp

def pt(Y):
 return((Y[0],Y[1],Y[2]))


a,b=2,3
v1=sp.Matrix([-1,1,-2])
v2=sp.Matrix([2,0,3])

v=a*v1+b*v2

i=sp.Matrix([1,0,0])
k=sp.Matrix([0,0,1])
j=sp.Matrix([0,1,0])
v3=v1+v2

\end{pycode} 
\begin{defin}
Um vetor ${\color{blue}\vec{b}}\in \R^n$ é uma {\color{blue} combinação linear} de vetores $\vec{v}_1, \vec{v}_2, \ldots, \vec{v}_k$ se existem escalares {\color{red}$\alpha_1,\alpha_2,\ldots, \alpha_k$} tais que
\[{\color{red}\alpha_1}\vec{v}_1+{\color{red}\alpha_2}\vec{v}_2+\cdots +{\color{red}\alpha_k}\vec{v}_k={\color{blue}\vec{b}}.\]
\end{defin}

\begin{exe}
Podemos escrever  {\color{blue}\(\vec{b}=\pyl{pt(v)}\)} como combinação linear da seguinte forma:
\begin{multicols}{2}
\begin{enumerate}[a]
\item \({\color{red}\pyl{v[0]}}\pyl{i}+{\color{red}\pyl{v[1]}}\pyl{j} +{\color{red}\pyl{v[2]}}\pyl{k}={\color{blue}\pyl{v}}\)
\item \({\color{red}\pyl{a}}\pyl{v1}+{\color{red}\pyl{b}}\pyl{v2}={\color{blue}\pyl{v}}\)
\end{enumerate}
\end{multicols}
\smallskip
\end{exe}

\end{frame}

\end{teo}

\subsection*{Dependência e Independência Linear}
%-----------------------------------------------------------------
\begin{frame}[label=lild]{Dependência e Independência Linear}
No item 4 deste \hyperlink{exem.comb}{\beamerbutton{Exemplo}}, vimos que o vetor $\vec{b}$ pode ser escrito como combinação apenas de \(\vec{v}_1\) e \(\vec{v}_2\) dispensando a necessidade do vetor \(\vec{v}_3\). Isso ocorre pois o vetor \(\vec{v}_3\) já é, por si só, uma combinação linear dos vetores \(\vec{v}_1\) e \(\vec{v}_2\). Isso motiva a seguinte definição.

\begin{defin}
Dizemos que um conjunto $W=\{\vec{v}_1,\vec{v}_2,\ldots,\vec{v}_k\}$ de vetores do $\R^n$ é {\color{blue} linearmente independente (LI)} se a equação vetorial
\[{\color{blue}\alpha_1}\vec{v}_1+{\color{blue}\alpha_2}\vec{v}_2+\cdots +{\color{blue}\alpha_k}\vec{v}_k=\vec{0}\]
{\color{red}só possui a solução trivial}. Em outras palavras, a única forma de escrever o vetor nulo como combinação linear dos vetores $\vec{v}_1,\vec{v}_2,\ldots, \vec{v}_k$ é aquela em que todos os escalares são iguais a zero. Caso contrário, dizemos que o conjunto $S$ é {\color{blue}linearmente dependente (LD)}.
\end{defin}
\end{frame}

\begin{frame}[label=lild]{}
\begin{exe}
Decida se os seguintes conjuntos de vetores são LI ou LD:
\begin{enumerate}
\item 
$\vec{v}_1=
\begin{bmatrix}
1 \\ 0 \\1
\end{bmatrix}$, 
$\vec{v}_2\begin{bmatrix}
0\\1\\1
\end{bmatrix}$, $\vec{v}_3=\begin{bmatrix}
1\\1\\1
\end{bmatrix}$.

\item $\vec{v}_1=
\begin{bmatrix}
1 \\ 0 \\1\\2
\end{bmatrix}$, 
$\vec{v}_2\begin{bmatrix}
2 \\ -1 \\ 3 \\ 5
\end{bmatrix}$, $\vec{v}_3=\begin{bmatrix}
-1 \\ 2\\-3\\-4
\end{bmatrix}$.
\end{enumerate}
\end{exe}

\begin{block}{}
Mostrar que um conjunto $S=\{\vec{v}_1,\vec{v}_2,\ldots,\vec{v}_k\}$ de vetores do $\R^n$ é {\color{red}LI} é equivalente a mostrar que o sistema {\color{red}$AX=0$ tem solução única}, onde $A$ é a matriz cujas {\color{red}colunas são dadas pelos vetores de $S$}.
\end{block}

\end{frame}

\begin{frame}[label=lild,fragile]{Solução do Exercício}
\begin{pycode} 
import sympy as sp

v1=sp.Matrix([1,2,-1,-2])
v2=sp.Matrix([0,1,2,5])
v3=sp.Matrix([1,0,-4,-1])
v4=sp.Matrix([1,-1,2,7])

A=sp.Matrix.hstack(v1,v2,v3,v4)

pos,neg=sp.symbols('+ -')
def sinal(x):
 if x>0:
  return pos
 else:
  return neg

def opa(M,i,j):
 B=M.elementary_row_op(op='n<->m', row=None, k=None, row1=i-1, row2=j-1)
 L=r'L_{}'.format(i)+r'\leftrightarrow L_{}'.format(j)
 return B, L

def opb(M,i,r):
 rr=sp.Abs(r)
 if rr==1:
  rr=' '
 B=M.elementary_row_op(op='n->kn', row=i-1, k=r, row1=None, row2=None)
 L=r'L_{}'.format(i)+r'\to '+r'{}'.format(sinal(r))+r'{}'.format(rr)+r'L_{}'.format(i)
 return B, L

def opc(M,i,r,j):
 rr=sp.Abs(r)
 if rr==1:
  rr=' '
 B=M.elementary_row_op(op='n->n+km', row=i-1, k=r, row1=None, row2=j-1)
 L=r'L_{}'.format(i)+r'\to L_{}'.format(i)+r'{}'.format(sinal(r))+r'{}'.format(rr)+r'L_{}'.format(j)
 return B, L

#1ª eliminação
A1,e1a=opc(A,2,-2,1)
A1,e1b=opc(A1,3,1,1)
A1,e1c=opc(A1,4,2,1)

#2ª eliminação
A2,e2a=opc(A1,3,-2,2)
A2,e2b=opc(A2,4,-5,2)

#3ª eliminação
A3,e3a=opc(A2,4,-11,3)

\begin{frame}[label=lild]{Propriedades}

\begin{enumerate}
\item Um conjunto formado por um único vetor não nulo é LI.

\item Um conjunto com 2 vetores não nulos é LD se, e somente se, são múltiplos.

\item Um conjunto finito de vetores de $\R^n$ que contém o vetor nulo é LD.

\item Se $\{\vec{v}_1,\vec{v}_2,\ldots,\vec{v}_k\}$ é LD, qualquer conjunto que o contenha também é LD.

\item As colunas de uma matriz $A$ de ordem $n$ são LI se, e somente se, $\det(A)\neq 0$.

\item Em $\R^n$, um conjunto com mais de $n$ vetores é LD.

\item Um conjunto de vetores é LD se, e somente se, {\color{red} pelo menos um dos vetores} pode ser escrito como combinação linear dos demais.
\end{enumerate}


\end{frame}
